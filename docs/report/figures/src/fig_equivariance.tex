  \end{aligned}$}\\[5pt]

\begin{minipage}[t]{7.9cm}\centering
  \textbf{(e) why the perturbation cancels}\\[8pt]
  {\small $\begin{aligned}
    \hat R(XQ^{\top})&=\hat R(X)\,Q^{\top},\qquad R=Q^{\top}\\[2pt]
    \angle\big(\hat R(XQ^{\top}),R\big)&=\angle\big(\hat R(X)Q^{\top},\,IQ^{\top}\big)=\angle\big(\hat R(X),I\big)
  \end{aligned}$}\\[5pt]
  {\footnotesize the error depends on the assembled fragment only}
